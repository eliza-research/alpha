// Controlled Android ports. ROUTER_BODY is the unmodified production touch path.
open class View {
    var parent: Any? = null
    var visibility = VISIBLE
    var left = 0
    var top = 0
    var translationX = 0f
    var translationY = 0f
    var importantForAccessibility = 0
    fun setWillNotDraw(value: Boolean) {}
    open fun dispatchTouchEvent(event: MotionEvent) = false
    companion object { const val VISIBLE = 0; const val IMPORTANT_FOR_ACCESSIBILITY_NO = 2 }
}
class ViewGroup : View() {
    val children = mutableListOf<View>()
    fun indexOfChild(view: View) = children.indexOf(view)
    fun add(view: View) { children.add(view); view.parent = this }
}
class MotionEvent(var action: Int, var x: Float, var y: Float, val downTime: Long) {
    val actionMasked get() = action
    fun offsetLocation(dx: Float, dy: Float) { x += dx; y += dy }
    fun recycle() {}
    companion object {
        const val ACTION_DOWN = 0; const val ACTION_UP = 1; const val ACTION_MOVE = 2; const val ACTION_CANCEL = 3
        fun obtain(event: MotionEvent) = MotionEvent(event.action, event.x, event.y, event.downTime)
    }
}
class OccludingSurfaceLayout : View() {
    var rects = false
    var selected = false
    var paused = false
    var disposed = false
    val delivered = mutableListOf<Int>()
    // Geometry is a controlled input; production rounded geometry has separate native tests.
    fun pageOwns(x: Float, y: Float) = rects && x >= 20f && x < 100f && y >= 0f && y < 100f
    override fun dispatchTouchEvent(event: MotionEvent): Boolean { delivered.add(event.action); return true }
}
// LAYER_BODY
class Router(private val pages: () -> List<OccludingSurfaceLayout>) : View() {
// ROUTER_BODY
    fun observedTarget(active: Boolean): String = when {
        !active -> "none"
        touchRouted && touchTarget != null -> "page"
        touchRouted -> "cancelled"
        else -> "host"
    }
}
class Fixture {
    val page = OccludingSurfaceLayout()
    val parent = ViewGroup()
    val router = Router { if (takesPageInput(SurfaceLayer.BELOW_HOST, page.selected, page.paused, page.disposed)) listOf(page) else emptyList() }
    var active = false
    var downTime = 1L
    init { parent.add(page); parent.add(router) }
    fun present(valid: Boolean) { page.selected = valid; page.rects = valid && !page.paused; page.visibility = if (page.selected && !page.paused) 0 else 8 }
    fun down(host: Boolean = false, inside: Boolean = true) { active = true; router.dispatchTouchEvent(MotionEvent(0, if (!inside) 120f else if (host) 10f else 50f, 50f, downTime)) }
    fun move() { router.dispatchTouchEvent(MotionEvent(2, 50f, 50f, downTime)) }
    fun secondFinger() { router.dispatchTouchEvent(MotionEvent(0, 50f, 50f, downTime)) }
    fun pageSecondFinger() { router.dispatchTouchEvent(MotionEvent(5, 10f, 50f, downTime)) }
    fun hide() { page.selected = false; page.rects = false; page.visibility = 8 }
    fun pause() { page.paused = true; page.rects = false; page.visibility = 8 }
    fun resume() { page.paused = false; page.visibility = if (page.selected) 0 else 8 }
    fun dispose() { page.disposed = true; hide() }
    fun up() { router.dispatchTouchEvent(MotionEvent(1, 50f, 50f, downTime)); active = false; downTime++ }
    fun observe() = "{\"s.target\":\"${router.observedTarget(active)}\"}"
}
fun main() {
    val traces = mutableListOf<String>()
    fun scenario(name: String, body: (Fixture, MutableList<String>) -> Unit) {
        val f = Fixture(); val steps = mutableListOf<String>(); body(f, steps)
        traces.add("{\"name\":\"$name\",\"expect\":{\"s.target\":\"none\"},\"steps\":[${steps.joinToString(",")}]} ")
    }
    fun step(f: Fixture, steps: MutableList<String>, actions: String) { steps.add("{\"action\":$actions,\"expect\":${f.observe()}}") }
    val present = "{\"name\":\"present\",\"args\":[true]}"
    val down = "{\"name\":\"down\",\"args\":[false,true]}"
    scenario("missing geometry refuses page input") { f, s -> f.down(); step(f,s,down) }
    scenario("outside page stays on host") { f, s ->
        f.present(true); step(f,s,present); f.down(inside = false); step(f,s,"{\"name\":\"down\",\"args\":[false,false]}")
    }
    scenario("host split finger stays on host") { f, s ->
        f.present(true); step(f,s,present)
        f.down(true); step(f,s,"{\"name\":\"down\",\"args\":[true,true]}")
        f.secondFinger(); step(f,s,"{\"name\":\"secondFinger\"}")
        f.move(); step(f,s,"{\"name\":\"move\"}")
    }
    scenario("page gesture retains ownership") { f, s ->
        f.present(true); step(f,s,present); f.down(); step(f,s,down)
        f.pageSecondFinger(); step(f,s,"{\"name\":\"secondFinger\"}")
        f.move(); step(f,s,"{\"name\":\"move\"}")
        f.up(); step(f,s,"{\"name\":\"up\"}")
    }
    for (withdraw in listOf("hide", "pause", "dispose")) scenario("$withdraw cancels and consumes the page gesture") { f, s ->
        f.present(true); step(f,s,present); f.down(); step(f,s,down)
        when(withdraw) { "hide" -> f.hide(); "pause" -> f.pause(); else -> f.dispose() }
        f.move(); step(f,s,"[{\"name\":\"$withdraw\"},{\"name\":\"move\"}]")
        check(f.page.delivered == listOf(0,3)) { "A withdrawn page must get one CANCEL" }
        f.move(); step(f,s,"{\"name\":\"move\"}")
        check(f.page.delivered == listOf(0,3)) { "Cancellation must not be replayed" }
        f.up(); step(f,s,"{\"name\":\"up\"}")
    }
    scenario("resume needs fresh geometry") { f, s ->
        f.present(true); step(f,s,present); f.pause(); step(f,s,"{\"name\":\"pause\"}")
        f.resume(); step(f,s,"{\"name\":\"resume\"}"); f.down(); step(f,s,down)
    }
    println("[${traces.joinToString(",")}]")
}
